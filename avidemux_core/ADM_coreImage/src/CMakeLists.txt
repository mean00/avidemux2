SET(ADM_LIB ADM_coreImage6)
ADM_ADD_SHARED_LIBRARY(${ADM_LIB})
TARGET_SOURCES(${ADM_LIB} PRIVATE
  ADM_image.cpp
  ADM_imageUtils.cpp
  ADM_imageResizer.cpp
  ADM_colorspace.cpp
  ADM_print.cpp
  ADM_imageSave.cpp
  ADM_imageOperation.cpp
  ADM_toneMapper.cpp
)
IF(USE_LIBPOSTPROC)
  TARGET_SOURCES(${ADM_LIB} PRIVATE ADM_pp.cpp)
ENDIF()
YASMIFY(bins ADM_imageUtils_asm)
TARGET_SOURCES(${ADM_LIB} PRIVATE ${bins})

TARGET_LINK_LIBRARIES(${ADM_LIB} PUBLIC ADM_core6 ADM_coreUtils6)
TARGET_LINK_LIBRARIES(${ADM_LIB} PUBLIC ADM_libavcodec ADM_libavutil ADM_libswscale)
IF(USE_LIBPOSTPROC)
  TARGET_LINK_LIBRARIES(${ADM_LIB} PUBLIC ADM_libpostproc)
ENDIF()
TARGET_INCLUDE_DIRECTORIES(${ADM_LIB} PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/../include)
